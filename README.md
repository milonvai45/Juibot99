   ❖ DEPLOY_WORKFLOWS ❖

```  
name: Node.js CI  
  
on:  
  push:  
    branches: [main]  
  pull_request:  
    branches: [main]  
  
jobs:  
  build:  
    runs-on: ubuntu-latest  
  
    strategy:  
      matrix:  
        node-version: [20.x]  
  
    steps:  
    - uses: actions/checkout@v2  
  
    - name: Use Node.js ${{ matrix.node-version }}  
      uses: actions/setup-node@v2  
      with:  
        node-version: ${{ matrix.node-version }}  
  
    - name: Install dependencies  
      run: npm install  
  
    - name: Start the bot  
      env:  
        PORT: 8080  
      run: npm start  
```
